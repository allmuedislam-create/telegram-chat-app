    <script>
        const INITIAL_BOTS = [
            { id: "gemini_ai", username: "Gemini AI", email: "gemini@ai.assistant", bio: "Your smart AI buddy powered by Gemini technology.", avatar: "", isBot: true, status: "bot" }
        ];

        const EMOJIS = ["😀", "😂", "🥰", "👍", "🔥", "🎉", "❤️", "🙌", "😭", "😮", "🙏", "✨"];

        const Storage = {
            getUsers: () => JSON.parse(localStorage.getItem('tg_users')) || [],
            setUsers: (users) => localStorage.setItem('tg_users', JSON.stringify(users)),
            getMessages: () => JSON.parse(localStorage.getItem('tg_messages')) || [],
            setMessages: (msgs) => localStorage.setItem('tg_messages', JSON.stringify(msgs)),
            getSession: () => JSON.parse(localStorage.getItem('tg_session')) || null,
            setSession: (session) => localStorage.setItem('tg_session', JSON.stringify(session)),
            getTheme: () => localStorage.getItem('tg_theme') || 'dark',
            setTheme: (theme) => localStorage.setItem('tg_theme', theme)
        };

        let currentUser = null;
        let activeChatUserId = null;
        let selectedMessageId = null;

        const Auth = {
            toggleView: (showLogin) => {
                document.getElementById('auth-view-login').style.display = showLogin ? 'block' : 'none';
                document.getElementById('auth-view-register').style.display = showLogin ? 'none' : 'block';
            },
            init: () => {
                const session = Storage.getSession();
                if (session) {
                    currentUser = session;
                    document.getElementById('auth-screen').style.display = 'none';
                    App.initDashboard();
                }
            },
            register: () => {
                const userNode = document.getElementById('reg-username').value.trim();
                const emailNode = document.getElementById('reg-email').value.trim();
                const passNode = document.getElementById('reg-password').value.trim();

                if (!userNode || !emailNode || !passNode) return alert('Fill out all inputs.');
                
                let users = Storage.getUsers();
                if (users.some(u => u.email.toLowerCase() === emailNode.toLowerCase())) {
                    return alert('Email address registered already.');
                }

                const newUser = {
                    id: 'u_' + Date.now(),
                    username: userNode,
                    email: emailNode,
                    password: passNode,
                    bio: 'Hey there! I am using Telegram.',
                    avatar: '',
                    isBot: false,
                    status: 'online'
                };

                users.push(newUser);
                Storage.setUsers(users);
                currentUser = newUser;
                Storage.setSession(newUser);
                
                document.getElementById('auth-screen').style.display = 'none';
                App.initDashboard();
            },
            login: () => {
                const emailNode = document.getElementById('login-email').value.trim();
                const passNode = document.getElementById('login-password').value.trim();

                if (!emailNode || !passNode) return alert('Fill out all fields.');

                const users = Storage.getUsers();
                const user = users.find(u => u.email.toLowerCase() === emailNode.toLowerCase() && u.password === passNode);

                if (!user) return alert('Invalid email or password parameters.');

                currentUser = user;
                Storage.setSession(user);
                document.getElementById('auth-screen').style.display = 'none';
                App.initDashboard();
            },
            logout: () => {
                localStorage.removeItem('tg_session');
                location.reload();
            }
        };

        const App = {
            initDashboard: () => {
                UI.applyTheme(Storage.getTheme());
                UI.buildEmojiPicker();
                App.syncUserDatabaseWithBots();
                App.renderUserList();
                UI.updateDrawerInfo();
                
                setInterval(() => {
                    if (activeChatUserId) App.renderMessages(false);
                }, 3000);

                window.addEventListener('click', () => {
                    document.getElementById('msg-context-menu').style.display = 'none';
                    document.getElementById('chat-header-menu').style.display = 'none';
                });
            },
            syncUserDatabaseWithBots: () => {
                let users = Storage.getUsers();
                INITIAL_BOTS.forEach(bot => {
                    if (!users.some(u => u.id === bot.id)) {
                        users.push(bot);
                    }
                });
                Storage.setUsers(users);
            },
            renderUserList: () => {
                const query = document.getElementById('user-search').value.toLowerCase();
                const users = Storage.getUsers().filter(u => u.id !== currentUser.id);
                const allMessages = Storage.getMessages();
                const target = document.getElementById('contacts-target');
                target.innerHTML = '';

                const filtered = users.filter(u => u.username.toLowerCase().includes(query));

                filtered.forEach(u => {
                    const threadMsgs = allMessages.filter(m => 
                        (m.senderId === currentUser.id && m.receiverId === u.id) ||
                        (m.senderId === u.id && m.receiverId === currentUser.id)
                    );
                    
                    let lastMsgText = u.bio || 'No status bio';
                    let lastMsgTime = '';
                    
                    if (threadMsgs.length > 0) {
                        const lastM = threadMsgs[threadMsgs.length - 1];
                        lastMsgText = lastM.mediaType === 'image' ? '🖼️ Photo' : lastM.text;
                        lastMsgTime = lastM.time;
                    }

                    const item = document.createElement('div');
                    item.className = `chat-item ${activeChatUserId === u.id ? 'active' : ''}`;
                    item.onclick = () => App.selectChat(u.id);

                    const avatarMarkup = UI.getAvatarMarkup(u);

                    item.innerHTML = `
                        ${avatarMarkup}
                        <div class="chat-item-body">
                            <div class="chat-item-header">
                                <span class="chat-item-name">${u.username}</span>
                                <span class="chat-item-time">${lastMsgTime}</span>
                            </div>
                            <div class="chat-item-msg">${lastMsgText}</div>
                        </div>
                    `;
                    target.appendChild(item);
                });
            },
            selectChat: (userId) => {
                activeChatUserId = userId;
                document.getElementById('app-viewport').classList.add('chat-open');
                
                const users = Storage.getUsers();
                const counterpart = users.find(u => u.id === userId);
                
                const headerNode = document.getElementById('chat-header-node');
                const inputNode = document.getElementById('chat-input-node');
                
                headerNode.style.display = 'flex';
                inputNode.style.display = 'flex';

                document.getElementById('header-name').innerText = counterpart.username;
                document.getElementById('header-status').innerText = counterpart.isBot ? 'bot assistant' : counterpart.status || 'online';
                
                const headAvContainer = document.getElementById('header-avatar');
                headAvContainer.className = `avatar ${counterpart.status === 'online' && !counterpart.isBot ? 'online' : ''}`;
                headAvContainer.style.background = counterpart.avatar ? 'none' : 'var(--tg-primary)';
                headAvContainer.innerHTML = counterpart.avatar ? `<img src="${counterpart.avatar}" class="avatar">` : UI.getInitials(counterpart.username);

                App.renderUserList();
                App.renderMessages(true);
            },
            renderMessages: (shouldScroll) => {
                if (!activeChatUserId) return;
                const bodyNode = document.getElementById('chat-body-node');
                const allMessages = Storage.getMessages();
                
                const threadMsgs = allMessages.filter(m => 
                    (m.senderId === currentUser.id && m.receiverId === activeChatUserId) ||
                    (m.senderId === activeChatUserId && m.receiverId === currentUser.id)
                );

                const currentHTML = bodyNode.innerHTML;
                let newHTML = '';

                if (threadMsgs.length === 0) {
                    newHTML = `<div class="no-chat-selected"><div class="no-chat-bubble">No messages here yet. Say hello!</div></div>`;
                } else {
                    threadMsgs.forEach(m => {
                        const isOut = m.senderId === currentUser.id;
                        let contentMarkup = `<span class="message-text">${UI.escapeHTML(m.text)}</span>`;
                        if (m.mediaType === 'image') {
                            contentMarkup = `<div class="message-media"><img src="${m.mediaData}"></div>` + (m.text ? `<span class="message-text">${UI.escapeHTML(m.text)}</span>` : '');
                        }

                        newHTML += `
                            <div class="message-row ${isOut ? 'out' : 'in'}">
                                <div class="message-bubble" data-msg-id="${m.id}">
                                    <div class="message-actions" onclick="UI.showMsgContextMenu(event, '${m.id}')"><i class="fa-solid fa-angle-down"></i></div>
                                    ${contentMarkup}
                                    <div class="message-footer">
                                        <span>${m.time}</span>
                                        ${isOut ? '<i class="fa-solid fa-check-double" style="font-size:10px; color:#55aaff;"></i>' : ''}
                                    </div>
                                </div>
                            </div>
                        `;
                    });
                }

                if (bodyNode.querySelector('.typing-indicator')) {
                    newHTML += `
                        <div class="message-row in" id="typing-row">
                            <div class="typing-indicator">
                                <div class="typing-dot"></div>
                                <div class="typing-dot"></div>
                                <div class="typing-dot"></div>
                            </div>
                        </div>
                    `;
                }

                if (currentHTML !== newHTML) {
                    const cachedTyping = bodyNode.querySelector('#typing-row');
                    bodyNode.innerHTML = newHTML;
                    if (cachedTyping && !bodyNode.querySelector('#typing-row')) {
                        bodyNode.appendChild(cachedTyping);
                    }
                    if (shouldScroll) UI.scrollToBottom();
                }
            },
            sendMessage: () => {
                const input = document.getElementById('msg-input');
                const text = input.value.trim();
                if (!text) return;

                App.commitMessageRecord(text, null, null);
                input.value = '';
                UI.autoGrow(input);
                App.renderMessages(true);
                App.renderUserList();

                if (activeChatUserId === 'gemini_ai') {
                    App.triggerBotResponseResponse(text);
                }
            },
            commitMessageRecord: (text, mediaType, mediaData) => {
                const allMessages = Storage.getMessages();
                const now = new Date();
                const timeString = now.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit', hour12: false });
                
                const newMsg = {
                    id: 'm_' + Date.now() + Math.random().toString(36).substr(2, 4),
                    senderId: currentUser.id,
                    receiverId: activeChatUserId,
                    text: text,
                    mediaType: mediaType,
                    mediaData: mediaData,
                    time: timeString
                };

                allMessages.push(newMsg);
                Storage.setMessages(allMessages);
            },
            triggerBotResponseResponse: (userPrompt) => {
                UI.showTypingIndicator(true);
                
                setTimeout(() => {
                    UI.showTypingIndicator(false);
                    const allMessages = Storage.getMessages();
                    const now = new Date();
                    const timeString = now.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit', hour12: false });
                    
                    let replyText = `I received your message: "${userPrompt}". This is a client-side Gemini AI placeholder response. To enable full AI automation, connect this to the live Gemini API endpoint.`;
                    
                    const botMsg = {
                        id: 'm_' + Date.now(),
                        senderId: 'gemini_ai',
                        receiverId: currentUser.id,
                        text: replyText,
                        mediaType: null,
                        mediaData: null,
                        time: timeString
                    };

                    allMessages.push(botMsg);
                    Storage.setMessages(allMessages);
                    App.renderMessages(true);
                    App.renderUserList();
                }, 1500);
            },
            deleteSelectedMessage: () => {
                if (!selectedMessageId) return;
                let allMessages = Storage.getMessages();
                allMessages = allMessages.filter(m => m.id !== selectedMessageId);
                Storage.setMessages(allMessages);
                App.renderMessages(false);
                App.renderUserList();
            },
            clearCurrentChat: () => {
                if (!activeChatUserId) return;
                let allMessages = Storage.getMessages();
                allMessages = allMessages.filter(m => 
                    !((m.senderId === currentUser.id && m.receiverId === activeChatUserId) ||
                      (m.senderId === activeChatUserId && m.receiverId === currentUser.id))
                );
                Storage.setMessages(allMessages);
                App.renderMessages(true);
                App.renderUserList();
            },
            saveProfileChanges: () => {
                const nextUser = document.getElementById('edit-username').value.trim();
                const nextBio = document.getElementById('edit-bio').value.trim();
                
                if (!nextUser) return alert('Username cannot be blank.');

                let users = Storage.getUsers();
                const index = users.findIndex(u => u.id === currentUser.id);
                
                if (index !== -1) {
                    users[index].username = nextUser;
                    users[index].bio = nextBio;
                    
                    const modalAv = document.getElementById('modal-avatar-preview');
                    const imgNode = modalAv.querySelector('img');
                    if (imgNode) {
                        users[index].avatar = imgNode.src;
                    }

                    currentUser = users[index];
                    Storage.setUsers(users);
                    Storage.setSession(currentUser);
                    
                    UI.updateDrawerInfo();
                    UI.closeModal('profile-modal');
                    App.renderUserList();
                    if (activeChatUserId) App.selectChat(activeChatUserId);
                }
            }
        };

        const UI = {
            getInitials: (name) => name ? name.charAt(0).toUpperCase() : 'U',
            getAvatarMarkup: (user) => {
                const stateClass = user.status === 'online' && !user.isBot ? 'online' : '';
                if (user.avatar) {
                    return `<div class="avatar ${stateClass}"><img src="${user.avatar}" class="avatar"></div>`;
                }
                return `<div class="avatar ${stateClass}" style="background: var(--tg-primary);">${UI.getInitials(user.username)}</div>`;
            },
            closeChat: () => {
                activeChatUserId = null;
                document.getElementById('app-viewport').classList.remove('chat-open');
                App.renderUserList();
            },
            autoGrow: (element) => {
                element.style.height = "24px";
                element.style.height = (element.scrollHeight - 12) + "px";
            },
            handleTextareaKey: (e) => {
                if (e.key === 'Enter' && !e.shiftKey) {
                    e.preventDefault();
                    App.sendMessage();
                }
            },
            scrollToBottom: () => {
                const bodyNode = document.getElementById('chat-body-node');
                bodyNode.scrollTop = bodyNode.scrollHeight;
            },
            toggleDrawer: (open) => {
                document.getElementById('app-drawer').classList.toggle('open', open);
                document.getElementById('drawer-backdrop').classList.toggle('open', open);
            },
            updateDrawerInfo: () => {
                document.getElementById('drawer-name-node').innerText = currentUser.username;
                document.getElementById('drawer-email-node').innerText = currentUser.email;
                const dAv = document.getElementById('drawer-avatar-node');
                dAv.style.background = currentUser.avatar ? 'none' : 'var(--tg-primary)';
                dAv.innerHTML = currentUser.avatar ? `<img src="${currentUser.avatar}" class="avatar" style="width:64px; height:64px;">` : UI.getInitials(currentUser.username);
            },
            toggleTheme: () => {
                const current = Storage.getTheme();
                const target = current === 'dark' ? 'light' : 'dark';
                UI.applyTheme(target);
            },
            applyTheme: (theme) => {
                document.documentElement.setAttribute('data-theme', theme);
                Storage.setTheme(theme);
                const darkMenuNode = document.querySelector('.drawer-menu-item i.fa-moon');
                if (darkMenuNode) {
                    darkMenuNode.parentElement.innerHTML = theme === 'dark' ? '<i class="fa-regular fa-sun"></i> Light Mode' : '<i class="fa-regular fa-moon"></i> Dark Mode';
                }
            },
            openProfileModal: () => {
                UI.toggleDrawer(false);
                document.getElementById('edit-username').value = currentUser.username;
                document.getElementById('edit-bio').value = currentUser.bio || '';
                const mAv = document.getElementById('modal-avatar-preview');
                mAv.style.background = currentUser.avatar ? 'none' : 'var(--tg-primary)';
                mAv.innerHTML = currentUser.avatar ? `<img src="${currentUser.avatar}" class="avatar" style="width:80px; height:80px;">` : UI.getInitials(currentUser.username);
                document.getElementById('profile-modal').classList.add('open');
            },
            closeModal: (id) => {
                document.getElementById(id).classList.remove('open');
            },
            openClearChatsModal: () => {
                UI.toggleDrawer(false);
                if (confirm('Are you sure you want to delete all conversation message logs?')) {
                    Storage.setMessages([]);
                    App.renderMessages(true);
                    App.renderUserList();
                    alert('All chat logs cleared.');
                }
            },
            triggerAvatarUpload: () => document.getElementById('file-input').click(),
            handleAvatarFile: (e) => {
                const file = e.target.files[0];
                if (!file) return;
                const reader = new FileReader();
                reader.onloadend = () => {
                    const mAv = document.getElementById('modal-avatar-preview');
                    mAv.style.background = 'none';
                    mAv.innerHTML = `<img src="${reader.result}" class="avatar" style="width:80px; height:80px;">`;
                }
                reader.readAsDataURL(file);
            },
            triggerMediaUpload: () => document.getElementById('media-input').click(),
            handleMediaFile: (e) => {
                const file = e.target.files[0];
                if (!file) return;
                const reader = new FileReader();
                reader.onloadend = () => {
                    App.commitMessageRecord("", "image", reader.result);
                    App.renderMessages(true);
                    App.renderUserList();
                    if (activeChatUserId === 'gemini_ai') {
                        App.triggerBotResponseResponse("Sent an image asset.");
                    }
                }
                reader.readAsDataURL(file);
            },
            buildEmojiPicker: () => {
                const popover = document.getElementById('emoji-popover-node');
                popover.innerHTML = '';
                EMOJIS.forEach(emo => {
                    const span = document.createElement('span');
                    span.className = 'emoji-item';
                    span.innerText = emo;
                    span.onclick = (e) => {
                        e.stopPropagation();
                        const input = document.getElementById('msg-input');
                        input.value += emo;
                        popover.classList.remove('open');
                        input.focus();
                    };
                    popover.appendChild(span);
                });
            },
            toggleEmojiPicker: (e) => {
                e.stopPropagation();
                document.getElementById('emoji-popover-node').classList.toggle('open');
            },
            showMsgContextMenu: (e, msgId) => {
                e.stopPropagation();
                selectedMessageId = msgId;
                const menu = document.getElementById('msg-context-menu');
                menu.style.top = `${e.clientY}px`;
                menu.style.left = `${e.clientX - 120}px`;
                menu.style.display = 'block';
            },
            toggleChatMenu: (e) => {
                e.stopPropagation();
                const menu = document.getElementById('chat-header-menu');
                menu.style.top = `${e.clientY + 15}px`;
                menu.style.left = `${e.clientX - 160}px`;
                menu.style.display = menu.style.display === 'block' ? 'none' : 'block';
            },
            showTypingIndicator: (show) => {
                const bodyNode = document.getElementById('chat-body-node');
                const exists = document.getElementById('typing-row');
                
                if (show && !exists) {
                    const row = document.createElement('div');
                    row.className = 'message-row in';
                    row.id = 'typing-row';
                    row.innerHTML = `
                        <div class="typing-indicator">
                            <div class="typing-dot"></div>
                            <div class="typing-dot"></div>
                            <div class="typing-dot"></div>
                        </div>
                    `;
                    bodyNode.appendChild(row);
                    UI.scrollToBottom();
                } else if (!show && exists) {
                    exists.remove();
                }
            },
            escapeHTML: (str) => {
                return str.replace(/[&<>'"]/g, 
                    tag => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', "'": '&#39;', '"': '&quot;' }[tag] || tag)
                );
            }
        };

        window.onload = () => {
            Auth.init();
            window.addEventListener('click', () => {
                document.getElementById('emoji-popover-node').classList.remove('open');
            });
        };
    </script>
